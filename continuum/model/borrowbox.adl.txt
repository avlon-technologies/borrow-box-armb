system BorrowBox — "A warm, trustworthy lending library: people list possessions, friends request dates, and both sides coordinate handoff, use and return."
  actor Borrower [person] | uses: ::BorrowBox — "A person who explores other people's items, requests dates and takes temporary custody. Any member may be a borrower of items they do not own."
  actor Owner [person] | uses: ::BorrowBox — "A person who lists their own items and decides who may borrow them and when. Authority over an item comes only from owning it."
  actor ProductTeam [team] | owns: ::BorrowBox, ::BorrowBox.Lending.Library — "The Avlon team accountable for BorrowBox and its lending rules."
  deployable WebApp — "One locally started process serving the interface and the API over one durable store."
  domain Lending (core) — "Temporary custody of privately owned items between people who know each other."
    bounded-context Library | realizes: ::BorrowBox.Lending.CatalogDiscovery, ::BorrowBox.Lending.ConditionEvidence, ::BorrowBox.Lending.ItemListing, ::BorrowBox.Lending.LoanCustody, ::BorrowBox.Lending.ReservationCoordination — "The authority for items, their reservations, loans, condition evidence and the history of who did what."
      business-rule CommittedReservationsNeverOverlap "A request may be approved only if its inclusive range overlaps no other approved, ready, borrowed or returning reservation of the same item; the check and the commit are one atomic act."
      business-rule EvidenceIsAppendOnly "Condition evidence is only ever added; an owner who disagrees with the borrower's account records their own evidence marked as a disagreement, and both remain."
      business-rule ExtensionRecheck "An extension may be accepted only while the loan is open and only if the days between the current and proposed due date overlap no other commitment for the item."
      business-rule HiddenItemsTakeNoNewRequests "A request to borrow a hidden item is refused; hiding changes nothing about requests and loans already made."
      business-rule LoanStepsInOrder "Ready follows approval, pickup follows ready, return starts only while borrowed, and confirmation follows the start of return."
      business-rule NoSelfBorrowing "A request to borrow is refused when the requester owns the item."
      business-rule OnlyPendingIsDecided "Approve and decline apply only to a pending request; cancel applies to the borrower's own request before pickup."
      business-rule OverdueIsInformational "A loan whose due date is before today in the community time zone is overdue; being overdue blocks nothing about returning it."
      business-rule RequestsFitTheCalendar "A requested range starts no earlier than today in the community time zone, ends on or after its start, and is no longer than the item's maximum loan length."
